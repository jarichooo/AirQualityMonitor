Code for the device (ESP32 S3 supermini)
#include <WiFi.h>
#include <PubSubClient.h>

// 1. Update these with your home Wi-Fi credentials
const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

// 2. Open Windows Command Prompt, type 'ipconfig', and find your IPv4 Address. Put it here.
const char* mqtt_server = "192.168.X.X"; 

WiFiClient espClient;
PubSubClient client(espClient);
const unsigned long publishInterval = 30000;
unsigned long lastPublish = 0;

void setup_wifi() {
  delay(10);
  Serial.print("Connecting to ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected.");
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection...");
    // Connect without a username/password since allow_anonymous is true in Mosquitto
    if (client.connect("ESP32_Supermini")) {
      Serial.println("connected");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" try again in 5 seconds");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, 1883);
  client.setBufferSize(512);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  unsigned long now = millis();
  if (now - lastPublish < publishInterval) {
    return;
  }
  lastPublish = now;

  // 3. Read your actual hardware sensors here (using dummy data for the example)
  float temp = 32.10;
  float hum = 65.50;
  float co2 = 415.0;
  float pm1 = 8.0;
  float pm10 = 22.0;
  float pm25 = 14.8;
  int mq135 = 15;
  int mq137 = 30;

  // 4. Format the JSON exactly as your Python ingest.py expects it
  char payload[300];
  snprintf(payload, sizeof(payload), 
    "{\"t\": %.2f, \"h\": %.2f, \"co2\": %.2f, \"pm1\": %.2f, \"pm25\": %.2f, \"pm10\": %.2f, \"mq135_raw\": %d, \"mq137_raw\": %d}",
    temp, hum, co2, pm1, pm25, pm10, mq135, mq137);

  // 5. Publish to the exact topicMosquitto is listening to
  Serial.print("Publishing message: ");
  Serial.println(payload);
  client.publish("poultry/sensors", payload);
}
